cmake_minimum_required(VERSION 3.24)
project(CODM_CAST_Studio VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Existing local CMake caches can contain empty release flag strings. Make the
# shipping configuration explicit for the app, core and bundled codecs alike.
if(MSVC)
 add_compile_options("$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Release>:/Ob2>")
 add_compile_definitions("$<$<CONFIG:Release>:NDEBUG>")
 add_link_options("$<$<CONFIG:Release>:/INCREMENTAL:NO>" "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>")
endif()
set(TEXTURE_DIR vendor/texture2ddecoder/src/Texture2DDecoder)
add_library(texture_decoder ${TEXTURE_DIR}/bcn.cpp ${TEXTURE_DIR}/etc.cpp ${TEXTURE_DIR}/astc.cpp ${TEXTURE_DIR}/atc.cpp ${TEXTURE_DIR}/pvrtc.cpp ${TEXTURE_DIR}/crunch.cpp ${TEXTURE_DIR}/unitycrunch.cpp)
if(MSVC)
 target_compile_options(texture_decoder PRIVATE /W0 /EHsc)
endif()
add_library(codm_core src/source.cpp src/model.cpp src/parts.cpp src/library.cpp src/materials.cpp src/uv_bake.cpp src/material_match.cpp src/textures.cpp src/animation.cpp src/rig.cpp src/animation_writer.cpp src/export.cpp src/cast_writer.cpp vendor/lz4/lz4.c vendor/sqlite/sqlite3.c)
target_sources(codm_core PRIVATE src/common.cpp src/legacy_animation.cpp src/animation_index.cpp src/export_actions.cpp src/animation_sources.cpp)
target_sources(codm_core PRIVATE src/effect_alpha.cpp)
target_sources(codm_core PRIVATE src/weapon_supplements.cpp)
target_sources(codm_core PRIVATE src/exported_animations.cpp)
target_sources(codm_core PRIVATE src/t6_candidate13.cpp)
target_sources(codm_core PRIVATE src/gallery.cpp)
target_sources(codm_core PRIVATE src/export_naming.cpp)
target_sources(codm_core PRIVATE src/batch_export.cpp)
target_sources(codm_core PRIVATE src/discovery.cpp)
target_sources(codm_core PRIVATE src/preview_pose.cpp)
target_sources(codm_core PRIVATE src/native_png.cpp)
target_sources(codm_core PRIVATE src/eye_material.cpp)
target_sources(codm_core PRIVATE src/model_pairs.cpp)
target_link_libraries(codm_core PUBLIC windowscodecs ole32)
target_link_libraries(codm_core PUBLIC texture_decoder bcrypt)
target_include_directories(codm_core PUBLIC src vendor)
target_compile_definitions(codm_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN GLM_ENABLE_EXPERIMENTAL SQLITE_THREADSAFE=1)
if(MSVC)
 target_compile_options(codm_core PUBLIC "$<$<COMPILE_LANGUAGE:CXX>:/W4;/EHsc;/utf-8;/bigobj>")
endif()
add_executable(codm_cli src/cli.cpp)
add_executable(codm_perf src/performance.cpp)
target_link_libraries(codm_perf PRIVATE codm_core)
set_target_properties(codm_perf PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
target_link_libraries(codm_cli PRIVATE codm_core)
set_target_properties(codm_cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
add_custom_command(TARGET codm_cli POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/data" "${CMAKE_SOURCE_DIR}/bin/data")
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/glfw EXCLUDE_FROM_ALL)
add_library(studio_imgui vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp vendor/imgui/misc/cpp/imgui_stdlib.cpp vendor/imgui/backends/imgui_impl_glfw.cpp vendor/imgui/backends/imgui_impl_opengl2.cpp)
target_include_directories(studio_imgui PUBLIC vendor/imgui)
target_link_libraries(studio_imgui PUBLIC glfw opengl32)
add_executable(CODM_CAST_Studio WIN32 src/studio.cpp)
set(CODM_STUDIO_OUTPUT_NAME "CODM_CAST_Studio" CACHE STRING "Studio executable filename without extension")
set_target_properties(CODM_CAST_Studio PROPERTIES OUTPUT_NAME "${CODM_STUDIO_OUTPUT_NAME}")
target_link_libraries(CODM_CAST_Studio PRIVATE codm_core studio_imgui ole32 shell32 uuid)
set_target_properties(CODM_CAST_Studio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
add_custom_command(TARGET CODM_CAST_Studio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/data" "${CMAKE_SOURCE_DIR}/bin/data")
enable_testing()
add_test(NAME core_self_test COMMAND codm_cli self-test)
add_executable(codm_core_tests src/core_tests.cpp)
target_link_libraries(codm_core_tests PRIVATE codm_core)
add_test(NAME native_regressions COMMAND codm_core_tests "${CMAKE_SOURCE_DIR}/work/tests")
add_test(NAME studio_state_checks COMMAND CODM_CAST_Studio --self-check "${CMAKE_SOURCE_DIR}/work/tests/studio-state.json")
add_test(NAME studio_input_checks COMMAND CODM_CAST_Studio --ui-check "${CMAKE_SOURCE_DIR}/work/tests/studio-input.json")
